'A versão atual precisa estar aprovada.' : !runAt ? 'Escolha data e hora ou use o botão Próximo horário às 5h.' : 'Horário selecionado: '+runAt.replace('T',' às ')+' (Brasília).'}</p>

                              const expectedAt=brasiliaToUtc(runAt), articleId=current.id, version=current.draft_version;
                              if(Date.parse(expectedAt)<=Date.now())throw new Error('Escolha uma data futura.');
                              const before:Schedule[]=await api('/schedules');
                              const matches=(s:Schedule)=>s.article_id===articleId && s.version===version && s.run_at===expectedAt && s.status==='queued';
                              if(!before.some(matches))await api('/articles/'+articleId+'/schedule','POST',{version,runAt:expectedAt});
                              const saved:Schedule[]=await api('/schedules');
                              if(!saved.some(matches))throw new Error('Não foi possível confirmar o registro na fila. Confira Agendamentos antes de tentar novamente.');

                              setNotice('Agendamento confirmado na fila: '+brasiliaDisplay(expectedAt)+' (Brasília), versão '+version+'.');
                            })}>{busy ? 'Verificando…' : 'Confirmar agendamento'}</button>
                            <button type="button" disabled={busy} onClick={()=>{setScheduleConfirm(false);setNotice('Agendamento não enviado.');}}>Voltar sem agendar</button>
